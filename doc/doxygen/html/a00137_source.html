168/*
169 * @brief Stop motor in case pf panic.
170 * This function stops the motor by resetting PA4, PA6 and PA7
171 * which are connected to PWM, INA and INB of the motor driver.
172 * It doesn't disengage the clutch.
173 * It is meant to be called by fault exceptions handlers.
174 * It is INLINE in order to use no function call nor stack.
175 */
176/* TODO test, make sure it is inlined, maybe use macro */
177__attribute__((always_inline)) inline void MOTOR_stopPanic(void)
178{
179 /* Reset PWN, INA and INB */
180 // LL_GPIO_ResetOutputPin(GPIOA,
181 // LL_GPIO_PIN_4 | LL_GPIO_PIN_6 | LL_GPIO_PIN_7);
182 /* Very low level : register acces */
183 GPIOA->BRR = LL_GPIO_PIN_4 | LL_GPIO_PIN_6 | LL_GPIO_PIN_7;
184}
